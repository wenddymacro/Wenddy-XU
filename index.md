<script>
function setLang(lang) {
  document.querySelectorAll('[data-lang]').forEach(function (el) {
    el.classList.toggle('show', el.getAttribute('data-lang') === lang);
  });
  document.getElementById('btn-zh').classList.toggle('active', lang === 'zh');
  document.getElementById('btn-en').classList.toggle('active', lang === 'en');
  try { localStorage.setItem('homepage-lang', lang); } catch (e) {}
}

(function initLang() {
  var lang = 'zh';
  try {
    lang = localStorage.getItem('homepage-lang') || ((navigator.language || '').toLowerCase().indexOf('zh') === 0 ? 'zh' : 'en');
  } catch (e) {}
  setLang(lang === 'en' ? 'en' : 'zh');
})();
</script>
